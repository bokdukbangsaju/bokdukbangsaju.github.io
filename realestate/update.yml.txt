name: Update Real Estate Dashboard

on:
  schedule:
    - cron: '0 0 1 * *'  # 매월 1일 자정 자동 실행 (UTC 기준)
  workflow_dispatch:      # GitHub 웹사이트에서 [Run workflow] 버튼으로 수동 실행 가능

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: 저장소 체크아웃
        uses: actions/checkout@v3

      - name: Python 환경 설정
        uses: actions/setup-python@v4
        with:
          python-version: '3.10'

      - name: 필수 라이브러리 설치
        run: |
          pip install pandas

      - name: 데이터 처리 파이썬 스크립트 실행
        run: |
          python busan_dashboard.py

      - name: 변경 사항 자동 커밋 및 푸시
        run: |
          git config --global user.name "github-actions[bot]"
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git add .
          git commit -m "자동 갱신: 최신 부동산 CSV 데이터 반영 [skip ci]" || exit 0
          git push
